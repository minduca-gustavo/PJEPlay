function observa(parametro) {
  const S = (window.__observa ??= { obs: null, log: [], inicio: null, onUnload: null });

  const desc = el => {
    let s = el.tagName.toLowerCase();
    if (el.id) s += '#' + el.id;
    if (typeof el.className === 'string' && el.className.trim())
      s += '.' + el.className.trim().split(/\s+/).join('.');
    return s;
  };
  const caminho = el => {
    const p = [];
    while (el && el.nodeType === 1 && p.length < 4) { p.unshift(desc(el)); el = el.parentElement; }
    return p.join(' > ');
  };
  const texto = n => {
    const t = (n.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 80);
    return t ? `"${t}"` : '';
  };
  const hora = () => {
    const d = new Date();
    return d.toLocaleTimeString('pt-BR', { hour12: false }) + '.' + String(d.getMilliseconds()).padStart(3, '0');
  };

  const baixar = motivo => {
    const txt = [
      `Página: ${location.href}`,
      `Início: ${S.inicio}`,
      `Fim: ${new Date().toLocaleString('pt-BR')} (${motivo})`,
      `Eventos: ${S.log.length}`,
      '',
      ...S.log
    ].join('\n');
    const a = document.createElement('a');
    a.href = URL.createObjectURL(new Blob([txt], { type: 'text/plain;charset=utf-8' }));
    a.download = `observa_${Date.now()}.txt`;
    a.click();
    setTimeout(() => URL.revokeObjectURL(a.href), 1000);
  };

  const parar = () => {
    S.obs?.disconnect();
    S.obs = null;
    if (S.onUnload) removeEventListener('beforeunload', S.onUnload);
    S.onUnload = null;
  };

  if (parametro === 'iniciar') {
    if (S.obs) return console.warn('observa: já está observando.');
    S.log = [];
    S.inicio = new Date().toLocaleString('pt-BR');

    S.obs = new MutationObserver(mutacoes => {
      for (const m of mutacoes) {
        const pai = caminho(m.target);
        m.addedNodes.forEach(n => {
          if (n.nodeType === 1) S.log.push(`[${hora()}] + ENTROU  ${pai} > ${desc(n)} ${texto(n)}`);
        });
        m.removedNodes.forEach(n => {
          if (n.nodeType === 1) S.log.push(`[${hora()}] - SAIU    ${pai} > ${desc(n)} ${texto(n)}`);
        });
      }
    });
    S.obs.observe(document.body, { childList: true, subtree: true });

    S.onUnload = () => {
      S.log.push(`[${hora()}] ### PÁGINA SENDO RECARREGADA/FECHADA`);
      parar();
      baixar('antes do recarregamento');
    };
    addEventListener('beforeunload', S.onUnload);

    console.log('observa: observando o body...');
  } else if (parametro === 'final') {
    if (!S.obs && !S.log.length) return console.warn('observa: nada foi observado.');
    parar();
    baixar('final manual');
    console.log(`observa: finalizado, ${S.log.length} eventos baixados.`);
  } else {
    console.warn("observa: use 'iniciar' ou 'final'.");
  }
}
